                </p>
                {jobStatus.results && typeof jobStatus.results === 'object' && 'statistics' in (jobStatus.results as Record<string, unknown>) && (
                  <div className="grid grid-cols-3 gap-4 pt-4 border-t border-border/50">
                    {(jobStatus.results as Record<string, unknown>).statistics && Object.entries((jobStatus.results as Record<string, Record<string, number>>).statistics as Record<string, number>)
                      .filter(([k]) => ['total_vehicles_detected', 'total_overspeed_events', 'avg_speed'].includes(k))
                      .map(([key, val]) => (
                        <div key={key} className="text-center">
                          <p className="text-lg font-bold">{typeof val === 'number' ? val.toFixed ? val.toFixed(1) : val : val}</p>
                          <p className="text-xs text-muted-foreground">{key.replace(/_/g, ' ').replace('total ', '')}</p>
                        </div>
                      ))
                    }
                  </div>
